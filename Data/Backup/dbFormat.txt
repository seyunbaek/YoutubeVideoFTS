CREATE TABLE "Channels" (
	"channelName"	TEXT,
	"channelID"	TEXT,
	"URL"	TEXT,
	"numVideos"	INTEGER,
	PRIMARY KEY("channelID")
)

CREATE TABLE "Videos" (
	"channel"	TEXT,
	"videoID"	TEXT,
	"thumbnailURL"	TEXT,
	"uploadDate"	TEXT,
	"transcribed"	INTEGER,
	"indexed"	INTEGER,
	"numLines"	INTEGER,
	PRIMARY KEY("videoID")
)

CREATE TABLE "Transcriptions" (
	"id"	INTEGER,
	"videoID"	TEXT,
	"startTime"	INTEGER,
	"endTime"	INTEGER,
	"text"	TEXT,
	"indexed"	INTEGER,
	PRIMARY KEY("id")
)
